#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

// Game config per difficulty
struct Difficulty {
    int maxNum;
    int maxAttempts;
    string name;
};

void playGame(Difficulty diff) {
    srand(time(0));
    int secret = rand() % diff.maxNum + 1;
    int attempts = diff.maxAttempts;
    int guess;

    cout << "\n=== " << diff.name << " Mode ===" << endl;
    cout << "Guess a number between 1 and "
         << diff.maxNum << endl;
    cout << "You have " << attempts
         << " attempts." << endl;

    while (attempts > 0) {
        cout << "\nAttempts left: " << attempts;
        cout << "\nYour guess: ";

        // Input validation
        if (!(cin >> guess)) {
            cout << "Invalid input. Numbers only!"
                 << endl;
            cin.clear();
            cin.ignore(1000, '\n');
            continue;
        }

        // Range check
        if (guess < 1 || guess > diff.maxNum) {
            cout << "Out of range! Enter 1 to "
                 << diff.maxNum << endl;
            continue;
        }

        attempts--;

        if (guess == secret) {
            cout << "\n🎉 Correct! You won in "
                 << (diff.maxAttempts - attempts)
                 << " guess(es)!" << endl;
            return;
        } else if (guess < secret) {
            cout << "Too low! Try higher." << endl;
        } else {
            cout << "Too high! Try lower." << endl;
        }
    }

    cout << "\n💀 Out of attempts! The number was "
         << secret << endl;
}

int main() {
    int choice;
    cout << "=== Number Guessing Game ===" << endl;
    cout << "1. Easy (1-50,  10 attempts)" << endl;
    cout << "2. Hard (1-100,  5 attempts)" << endl;
    cout << "0. Exit" << endl;

    Difficulty easy = {50,  10, "Easy"};
    Difficulty hard = {100,  5, "Hard"};

    while (true) {
        cout << "\nSelect difficulty: ";
        cin >> choice;

        if (choice == 0) {
            cout << "Thanks for playing!" << endl;
            break;
        } else if (choice == 1) {
            playGame(easy);
        } else if (choice == 2) {
            playGame(hard);
        } else {
            cout << "Invalid choice." << endl;
        }
    }
    return 0;
}
Code breakdown
🎲 rand() & srand()
Uses srand(time(0)) to seed the random number generator — ensures a different secret number every run, not a predictable sequence.
srand(time(0)) · rand() % maxNum + 1
⚡ Dynamic difficulty
Difficulty struct holds maxNum and maxAttempts — Easy gets 1–50 with 10 tries, Hard gets 1–100 with 5. One function handles both modes.
struct Difficulty { maxNum, maxAttempts }
🛡️ Input validation
Handles non-numeric input with cin.clear() and cin.ignore() — prevents runtime crashes on bad input. Range checked separately.
if (!(cin >> guess)) → clear + ignore
🔁 Game loop
while(attempts > 0) tracks remaining guesses — exits cleanly on win or loss, revealing the secret number if the player runs out.
while(attempts > 0) · return on win
